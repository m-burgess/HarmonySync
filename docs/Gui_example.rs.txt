use iced::widget::{button, column, container, row, scrollable, slider, text, text_input};
use iced::{Alignment, Element, Length, Sandbox, Settings, Theme};
use iced::theme::Palette;

#[derive(Default)]
struct FreeShowApp {
    current_slide: usize,
    slides: Vec<String>,
}

#[derive(Debug, Clone)]
enum Message {
    SelectSlide(usize),
    EditSlide(usize, String),
    VolumeChanged(f32),
    // Add more messages as needed
}

impl Sandbox for FreeShowApp {
    type Message = Message;

    fn new() -> Self {
        Self {
            slides: vec!["Slide 1".to_string(), "Slide 2".to_string(), "Slide 3".to_string()],
            current_slide: 0,
        }
    }

    fn title(&self) -> String {
        String::from("FreeShow Mimic - Iced GUI")
    }

    fn update(&mut self, message: Message) {
        match message {
            Message::SelectSlide(idx) => self.current_slide = idx,
            Message::EditSlide(idx, content) => self.slides[idx] = content,
            Message::VolumeChanged(val) => {
                // Handle volume
            }
        }
    }

    fn theme(&self) -> Theme {
        Theme::Dark
    }

    fn background_color(&self) -> Option<iced::Background> {
        Some(iced::Background::Color([0.1, 0.1, 0.1, 1.0].into()))
    }

    fn view(&self) -> Element<Message> {
        let content = row![
            // Column 1: Setlists (Scrollable list)
            container(
                scrollable(
                    column(
                        self.slides
                            .iter()
                            .enumerate()
                            .map(|(i, slide)| {
                                button(text(slide.clone()).size(16.0))
                                    .padding(8)
                                    .on_press(Message::SelectSlide(i))
                                    .into()
                            })
                            .collect::<Vec<_>>()
                    )
                    .spacing(4)
                )
                .width(Length::Fill)
            )
            .width(200.0)
            .height(Length::Fill)
            .style(|_| container::Appearance {
                background: Some(iced::Background::Color([0.15, 0.15, 0.15, 1.0].into())),
                ..Default::default()
            }),

            // Column 2: Slides (Editable text input for current slide)
            container(
                text_input("Edit slide...", &self.slides[self.current_slide])
                    .on_input(move |s| Message::EditSlide(self.current_slide, s))
                    .padding(16)
                    .size(20.0)
                    .width(Length::Fill)
                    .height(Length::Fill)
            )
            .width(300.0)
            .height(Length::Fill)
            .center_x()
            .center_y(),

            // Column 3: Preview (Text display of current slide)
            container(
                text(&self.slides[self.current_slide])
                    .size(32.0)
                    .horizontal_alignment(Alignment::Center)
            )
            .width(Length::Fill)
            .height(Length::Fill)
            .center_x()
            .center_y()
            .style(|_| container::Appearance {
                background: Some(iced::Background::Color([0.2, 0.2, 0.2, 1.0].into())),
                ..Default::default()
            })
        ]
        .spacing(10)
        .align_items(Alignment::Center);

        // Bottom: Multitrack Mixer (Reaper-like dark theme)
        let mixer = container(
            column![
                text("Multitrack Mixer").size(24.0),
                // Simulated tracks as rows
                row![
                    // Track 1
                    column![
                        text("Track 1"),
                        slider(0.0..=1.0, 0.5, Message::VolumeChanged).width(Length::Fixed(100.0)),
                        text("Mute").map(|_| button("Mute").into()),
                    ]
                    .spacing(4)
                    .width(150.0),
                    // Track 2
                    column![
                        text("Track 2"),
                        slider(0.0..=1.0, 0.5, Message::VolumeChanged).width(Length::Fixed(100.0)),
                        text("Mute").map(|_| button("Mute").into()),
                    ]
                    .spacing(4)
                    .width(150.0),
                    // Add more tracks...
                ]
                .spacing(10)
            ]
            .spacing(10)
            .padding(10)
        )
        .width(Length::Fill)
        .height(100.0)
        .style(|_| container::Appearance {
            background: Some(iced::Background::Color([0.05, 0.05, 0.05, 1.0].into())),
            ..Default::default()
        });

        column![content, mixer]
            .spacing(10)
            .padding(10)
            .into()
    }
}

fn main() -> iced::Result {
    FreeShowApp::run(Settings {
        window: iced::window::Settings {
            size: (1200, 800).into(),
            ..Default::default()
        },
        ..Default::default()
    })
}
